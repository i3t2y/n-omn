```js
const crypto = require('crypto');
const fs = require('fs');

const INTERNAL_PSK = process.env.INTERNAL_PSK;
if (!INTERNAL_PSK || INTERNAL_PSK.length < 16) {

function timingSafeEqual(a, b) {
  if (a.length !== b.length) return false;
  return crypto.timingSafeEqual(Buffer.from(a), Buffer.from(b));
}
function normalizePath(u) {
  try { const p = new URL(u, 'http://x').pathname; return p === '/' ? '/' : p.replace(/\/+$/, ''); }
  catch { return u; }
}

const diag = { start: Date.now(), errors: {}, recent: [] };
function rec(method, path, status, ms, len) {
  diag.recent.push({ ts: new Date().toISOString(), method, path, status, ms, len });
  if (diag.recent.length > 20) diag.recent.shift();
  if (status >= 500) diag.errors[status] = (diag.errors[status] || 0) + 1;
}

const server = http.createServer((req, res) => {
  const t0 = Date.now();
  const normPath = normalizePath(req.url);

  if (req.method === 'GET' && normPath === '/healthz') {
    const hc = http.request(
      { host: '127.0.0.1', port: OR_PORT, path: '/api/monitoring/health', method: 'GET', timeout: 5000 },
      (up) => {
        const ok = up.statusCode >= 200 && up.statusCode < 300;
        res.writeHead(ok ? 200 : 503, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify({ ok })); up.resume();

    hc.on('timeout', () => hc.destroy());
    hc.end(); return;
  }

    res.writeHead(200, { 'Content-Type': 'application/json' });

    return;

    res.writeHead(404, { 'Content-Type': 'application/json' });

    { host: '127.0.0.1', port: OR_PORT, path: req.url, method: req.method, headers: req.headers },

      res.writeHead(status, up.headers);
      if (isSSE) {
        up.on('data', (c) => { if (!res.write(c)) { up.pause(); res.once('drain', () => up.resume()); } });

    if (!res.headersSent) { res.writeHead(502, { 'Content-Type': 'application/json' }); }
    res.end(JSON.stringify({ error: 'bad_gateway', detail: err.message }));
  });

  // SSE/长思考：socket 不活跃超时全禁（首 token 前静默可达 120s+）
  proxyReq.setTimeout(0);
  req.socket.setTimeout(0);
  res.socket.setTimeout(0);

  req.on('aborted', () => { proxyReq.destroy(); });
  res.on('close', () => { if (!res.writableFinished) proxyReq.destroy(); });

  req.pipe(proxyReq);
});
```

**补丁 A：僵尸连接 GC（新增函数，在注册完 Key、Fetch provider IDs 之后调用）**

```bash
# ── 僵尸连接 GC：删除不在当前 NIM_KEYS 集合内的 nvidia 连接 ──
# 根因（日志C）：增量模式只增不删，Keys:7 但 Provider IDs:25，
#   18 个已移出 NIM_KEYS 的旧连接仍参与轮询， quota 被盗刷且拖低成功率。
gc_stale_nvidia_connections() {
  echo "[init] gc_stale: 扫描不在当前 NIM_KEYS 内的 nvidia 连接..."
  local _json _ids _id _name _del=0 _keep=0
  _json=$(curl -s -b "$COOKIE_FILE" "$BASE_URL/api/providers" 2>/dev/null || echo "")
  # 归一化：数组直取，对象取 .providers；逐项 {id,name}
  local _pairs
  _pairs=$(printf '%s' "$_json" | jq -r '
    (if type=="array" then . else (.providers // .data // []) end)
    | .[]? | select(type=="object")
    | select((.type // .provider // .slug // "") == "nvidia")
    | "\(.id)\t\(.name // .label // "")"' 2>/dev/null)
  # 当前 key 数 -> 合法名称集合 nim-01..nim-NN
  local _n; _n=$(printf '%s\n' "$NIM_KEYS" | sed '/^[[:space:]]*$/d' | wc -l)
  while IFS=$'\t' read -r _id _name; do
    [ -z "$_id" ] && continue
    if printf '%s' "$_name" | grep -qE '^nim-[0-9]+$'; then
      local _idx=$((10#$(printf '%s' "$_name" | sed 's/nim-0*//')))
      if [ "$_idx" -le "$_n" ]; then _keep=$((_keep+1)); continue; fi
    fi
    # 非 nim-NN 命名或编号超出当前 key 数 -> 删除
    local _c
    _c=$(curl -s -o /dev/null -w "%{http_code}" -b "$COOKIE_FILE" \
      -X DELETE "$BASE_URL/api/providers/$_id" 2>/dev/null || echo "000")
    echo "[init] gc_stale: DELETE $_name($_id) -> HTTP $_c"
    _del=$((_del+1))
  done <<< "$_pairs"
  echo "[init] gc_stale: 保留=$_keep 删除=$_del（当前 NIM_KEYS=$_n）"
}
```

```bash
# 在原有 3 表清扫后追加：全库文本列扫描 20129/127.0.0.1 残留
# 覆盖 api_keys.proxy_id、settings、provider_node_custom_paths、
# inspector_custom_hosts、free_proxies、relay_proxies 等 4.2.3 未扫面
deep_purge_proxy_residue() {
  [ "$_PURGE_PROXY" != "1" ] && return 0
  [ -f "$_DB_PATH" ] || return 0
  echo "[init] deep_purge: 全库扫描 20129 残留..."
  local _tables _t _cols _c _hits _total=0
  _tables=$(sqlite3 "$_DB_PATH" "SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%';" 2>/dev/null)
  while IFS= read -r _t; do
    [ -z "$_t" ] && continue
    _cols=$(sqlite3 "$_DB_PATH" "SELECT name FROM pragma_table_info('$_t') WHERE type IN ('TEXT','VARCHAR','CHAR') OR type LIKE '%TEXT%';" 2>/dev/null)
    [ -z "$_cols" ] && continue
    # 构造 OR 条件扫描所有文本列
    local _where="" _col
    while IFS= read -r _col; do
      [ -z "$_col" ] && continue
      [ -n "$_where" ] && _where="$_where OR "
      _where="$_where\"$_col\" LIKE '%20129%'"
    done <<< "$_cols"
    [ -z "$_where" ] && continue
    _hits=$(sqlite3 "$_DB_PATH" "SELECT COUNT(*) FROM \"$_t\" WHERE $_where;" 2>/dev/null || echo "0")
    if [ "${_hits:-0}" -gt 0 ]; then
      echo "[init] deep_purge: $_t 命中 $_hits 行 -> DELETE"
      sqlite3 "$_DB_PATH" "DELETE FROM \"$_t\" WHERE $_where;" 2>/dev/null || true
      _total=$((_total + _hits))
    fi
  done <<< "$_tables"
  # 补刀：api_keys.proxy_id 置空 + settings 内 proxy 键删除
  sqlite3 "$_DB_PATH" "UPDATE api_keys SET proxy_id=NULL WHERE proxy_id IS NOT NULL;" 2>/dev/null || true
```
