### **纠错 1:`20129` 根本不是"死代理",而是 OmniRoute 自己的 API 端口**


### **1. `init-nim-keys.sh`(核心修正版)**

```bash
#!/bin/bash
set -eo pipefail

# ─────────────────────────────────────────────────────────────
# NIM OmniRoute initializer v4.1.0 (源码校准版)
#   纠错依据:
#     - 20129 是 API_PORT，非代理 (Docker Guide)
#     - 代理循环根因 = proxy_registry 非空触发 auto-select 兜底 (Issue #3332)
#     - Combo 400 根因 = models 必须为对象数组 [{"model":"x"}] (User Guide)
# ─────────────────────────────────────────────────────────────

# --- [1] 单变量调试开关 ---
if [ "$NIM_MODE" = "DEBUG" ]; then
    LOG_DIR="/data/omni-data/log"
    mkdir -p "$LOG_DIR"
    exec > >(tee -a "$LOG_DIR/init_$(date +%Y%m%d_%H%M%S).log") 2>&1
    echo "[init] 🛠️ DEBUG 模式已激活，日志归档至 $LOG_DIR"
else
    LOG_DIR="/tmp"
fi

[ -z "$OMNIROUTE_PORT" ] && OMNIROUTE_PORT=20128
BASE_URL="http://127.0.0.1:$OMNIROUTE_PORT"
_DB_PATH="${DATA_DIR:-/data}/storage.sqlite"
COOKIE_FILE="/tmp/omniroute-cookie.txt"

# 稳定模型池（已剔除权限受限的 yi-large / kimi 等 404 模型）
STABLE_MODELS=("nvidia/z-ai/glm-5.2" "nvidia/deepseek-ai/deepseek-v4-pro" "nvidia/nvidia/nemotron-3-super-120b-a12b" "nvidia/meta/llama-3.2-90b-vision-instruct")

# --- [2] 根治代理循环：清空 registry + 关闭 proxy_enabled（Issue #3332）---
kill_proxy_root_cause() {
    echo "[init] 清理 proxy_registry 以关闭 auto-select 兜底 (Issue #3332)..."
    if [ -f "$_DB_PATH" ]; then
        sqlite3 "$_DB_PATH" "DELETE FROM proxy_registry;" 2>/dev/null || true
        sqlite3 "$_DB_PATH" "DELETE FROM proxy_assignments;" 2>/dev/null || true
        sqlite3 "$_DB_PATH" "UPDATE provider_connections SET proxy_enabled=0;" 2>/dev/null || true
        local _cnt
        _cnt=$(sqlite3 "$_DB_PATH" "SELECT COUNT(*) FROM proxy_registry;" 2>/dev/null || echo "?")
        echo "[init] proxy_registry 剩余记录: $_cnt (期望 0)"
    fi
}

# --- [3] Combo 创建：models 必须是对象数组 [{"model":"x"}] ---
create_combo() {
    local NAME="$1"; shift
    # 关键修正：生成 [{"model":"a"},{"model":"b"}]，而非 ["a","b"]
    local MODELS_JSON
    MODELS_JSON=$(printf '%s\n' "$@" | jq -R '{model: .}' | jq -s -c .)
    local BODY
    BODY=$(jq -n --arg n "$NAME" --argjson m "$MODELS_JSON" \
        '{name:$n, strategy:"priority", models:$m}')

    local RESP_FILE="$LOG_DIR/combo_${NAME}_resp.json"
    local C
    C=$(curl -s -o "$RESP_FILE" -w "%{http_code}" -b "$COOKIE_FILE" \
        -X POST "$BASE_URL/api/combos" -H "Content-Type: application/json" -d "$BODY")
    if [ "$C" = "200" ] || [ "$C" = "201" ]; then
        echo "[init] ✅ Combo $NAME 创建成功 (HTTP $C)"
    else
        echo "[init] ❌ Combo $NAME 失败 (HTTP $C)。响应:"; cat "$RESP_FILE"
    fi
}

# --- 主流程 ---
echo "[init] Starting v4.1.0..."
kill_proxy_root_cause

echo "[init] 等待 OmniRoute 就绪..."
until curl -sf "$BASE_URL/api/monitoring/health" >/dev/null 2>&1; do sleep 2; done

LOGIN_BODY=$(jq -n --arg password "$INITIAL_PASSWORD" '{password:$password}')
curl -s -o /dev/null -c "$COOKIE_FILE" -X POST "$BASE_URL/api/auth/login" \
    -H "Content-Type: application/json" -d "$LOGIN_BODY"

# 全局：清除任何 global 代理配置（DELETE 而非注入伪配置）
curl -s -o /dev/null -b "$COOKIE_FILE" -X DELETE "$BASE_URL/api/settings/proxy?level=global" 2>/dev/null || true

# 组合超时用官方 combo 默认（targetTimeoutSec 属于 Settings→Routing，不进 combo body）
curl -s -o /dev/null -b "$COOKIE_FILE" -X PATCH "$BASE_URL/api/settings" \
    -H "Content-Type: application/json" \
    -d '{"fallbackStrategy":"round-robin","requestRetry":1}'

# 注册后再清一次（防止注册流程重新写入 registry）
kill_proxy_root_cause
create_combo "nim-pool" "${STABLE_MODELS[@]}"

echo "[init] Done. v4.1.0"
```

```bash
purge_proxy_db() {
  [ "$_PURGE_PROXY" != "1" ] && { echo "[init] purge_proxy_db: skipped."; return 0; }
  local LIST_JSON
  LIST_JSON=$(curl -s -b "$COOKIE_FILE" "$BASE_URL/api/v1/management/proxies" 2>/dev/null || echo "")
  if [ -n "$LIST_JSON" ] && printf '%s' "$LIST_JSON" | jq -e . >/dev/null 2>&1; then
    local BAD_IDS
    BAD_IDS=$(printf '%s' "$LIST_JSON" | jq -r --arg h "$_PROXY_RELAY_HOST" --argjson p "$_PROXY_RELAY_PORT" \
      '(.proxies // .data // .) | (if type=="array" then . else [] end)
       | .[] | select((.host==$h) and ((.port|tonumber?)==$p)) | .id' 2>/dev/null)
    if [ -n "$BAD_IDS" ]; then
      local _id _c
      while IFS= read -r _id; do
        [ -z "$_id" ] && continue
        _c=$(curl -s -o /dev/null -w "%{http_code}" -b "$COOKIE_FILE" \
          -X DELETE "$BASE_URL/api/v1/management/proxies?id=${_id}&force=1" 2>/dev/null || echo "000")
        echo "[init] purge: API force-delete $_id -> HTTP $_c"
      done <<< "$BAD_IDS"
    else
      echo "[init] purge: 注册表无 ${_PROXY_RELAY_HOST}:${_PROXY_RELAY_PORT}。"
    fi
  else
    echo "[init] purge: 管理 API 暂不可用，走 SQL 兜底。"
  fi
  if [ -f "$_DB_PATH" ]; then
    sqlite3 "$_DB_PATH" "DELETE FROM proxy_assignments WHERE proxy_id IN
      (SELECT id FROM proxy_registry WHERE host='$(sql_escape "$_PROXY_RELAY_HOST")' AND port=$_PROXY_RELAY_PORT);" 2>/dev/null || true
    sqlite3 "$_DB_PATH" "UPDATE provider_connections SET proxy_enabled=0 WHERE provider='nvidia';" 2>/dev/null || true
    sqlite3 "$_DB_PATH" "DELETE FROM proxy_registry WHERE host='$(sql_escape "$_PROXY_RELAY_HOST")' AND port=$_PROXY_RELAY_PORT;" 2>/dev/null || true
    local _reg _asg _proxy_on
    _reg=$(sqlite3 "$_DB_PATH" "SELECT COUNT(*) FROM proxy_registry;" 2>/dev/null || echo "?")
    _asg=$(sqlite3 "$_DB_PATH" "SELECT COUNT(*) FROM proxy_assignments;" 2>/dev/null || echo "?")
    _proxy_on=$(sqlite3 "$_DB_PATH" "SELECT COUNT(*) FROM provider_connections WHERE provider='nvidia' AND proxy_enabled=1;" 2>/dev/null || echo "?")
    echo "[init] purge: registry=$_reg assignments=$_asg proxy_enabled=1剩余=$_proxy_on（期望 0/0/0）。"
  fi
}

# 探活：查 NVIDIA /v1/models，把目录里不存在的写进 deprecated

check_nim_model_health() {
  echo "[init] check_nim_model_health..."
  > /tmp/nim-deprecated.txt
  local _first_key _models_json _model_count
  _first_key=$(printf '%s\n' "$NIM_KEYS" | head -n1)
  _models_json=$(curl -s --max-time 10 -H "Authorization: Bearer ${_first_key}" \
    "https://integrate.api.nvidia.com/v1/models" 2>/dev/null || echo "")
  _model_count=$(printf '%s' "$_models_json" | jq -r '.data[]?.id' 2>/dev/null | wc -l)
  [ "${_model_count:-0}" -lt 5 ] && { echo "[init] only $_model_count models, skip 过滤"; return 0; }
  while IFS= read -r model; do
    [ -z "$model" ] && continue
    if ! printf '%s' "$_models_json" | jq -e --arg m "$model" 'any(.data[]?.id == $m; .)' >/dev/null 2>&1; then
      echo "[init]   $model — DEPRECATED（NVIDIA 目录无）"; echo "$model" >> /tmp/nim-deprecated.txt
    else
      [ "$NIM_MODE" = "DEBUG" ] && echo "[init]   $model — available"
    fi
  done < <(build_all_models)
  echo "[init] $(wc -l < /tmp/nim-deprecated.txt 2>/dev/null || echo 0) deprecated / $_model_count available"
}
```

3.  **模型 404**：原因是厂商前置（Prefix）缺失或模型目录更新。**修正**：按 2026-07-09 最新目录补全厂商前置，并引入 `check_nim_model_health` 实时过滤。

#### **运维指令**
*   **一键排查**：在 HF Settings 设置 `NIM_MODE=DEBUG`，然后执行 **Factory Reboot**。
*   **日志路径**：宿主机 `nomke/omni-data/log/` 下。
